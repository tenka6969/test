local function playSound(id, volume, pitch)

	s.SoundId = "rbxassetid://" .. tostring(id)
	s.Volume = volume or 0.5
	s.PlaybackSpeed = pitch or 1
	s.Parent = soundFolder
	s:Play()
	s.Ended:Connect(function()
		s:Destroy()
	end)

		}):Play()
		TweenService:Create(btnStroke, TweenInfo.new(0.2), {Transparency = 0.65}):Play()

-- 🌙 2. EYE-FRIENDLY NIGHT VISION (SMOOTH)
-- ==========================================
local function applyNightVision(enabled)
	if lightingTween then lightingTween:Cancel() end
	if nvTween then nvTween:Cancel() end

	if enabled then
		player:SetAttribute("MazeGuideActive", true)

		if not nvEffect then
			nvEffect = Instance.new("ColorCorrectionEffect")
			nvEffect.Name = "MazeNightVision"
			nvEffect.Brightness = 0
			nvEffect.Contrast = 0
			nvEffect.Saturation = 0
			nvEffect.TintColor = Color3.fromRGB(255, 255, 255)
			nvEffect.Parent = Lighting
		end

		-- Choose between Soft Cozy Glow (no eye strain!) or Clear Bright Glow
		local targetBrightness = (visionMode == "SOFT") and 1.6 or 2.4
		local targetAmbient = (visionMode == "SOFT") and Color3.fromRGB(115, 110, 138) or Color3.fromRGB(165, 165, 180)
		local targetCCBright = (visionMode == "SOFT") and 0.05 or 0.12
		local targetTint = (visionMode == "SOFT") and Color3.fromRGB(255, 246, 228) or Color3.fromRGB(245, 250, 255)

		lightingTween = TweenService:Create(Lighting, TweenInfo.new(1.0, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
			Brightness = targetBrightness,
			Ambient = targetAmbient,
			OutdoorAmbient = targetAmbient,
			FogEnd = 100000
		})
		lightingTween:Play()

		nvTween = TweenService:Create(nvEffect, TweenInfo.new(1.0, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
			Brightness = targetCCBright,
			Contrast = 0.08,
			Saturation = 0.12,
			TintColor = targetTint
		})
		nvTween:Play()

		if wispLight then
			TweenService:Create(wispLight, TweenInfo.new(0.8), {
				Brightness = (visionMode == "SOFT") and 1.3 or 2.2,
				Range = (visionMode == "SOFT") and 45 or 60
			}):Play()
		end
	else
		-- Smoothly fade back to the maze's original dark atmosphere
		lightingTween = TweenService:Create(Lighting, TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
			Brightness = savedLighting.Brightness,
			Ambient = savedLighting.Ambient,
			OutdoorAmbient = savedLighting.OutdoorAmbient,
			FogEnd = savedLighting.FogEnd
		})
		lightingTween:Play()
		Lighting.GlobalShadows = savedLighting.GlobalShadows

		if nvEffect then
			local oldNv = nvEffect
			nvEffect = nil
			local fadeOut = TweenService:Create(oldNv, TweenInfo.new(0.8), {
				Brightness = 0,
				Contrast = 0,
				Saturation = 0,
				TintColor = Color3.fromRGB(255, 255, 255)
			})
			fadeOut:Play()
			fadeOut.Completed:Connect(function()
				oldNv:Destroy()
			end)
		end

		player:SetAttribute("MazeGuideActive", false)
	end
end

visionBtn.MouseButton1Click:Connect(function()
	SFX.Click()
	if visionMode == "SOFT" then
		visionMode = "BRIGHT"
		visionBtn.Text = "☀️ Vision: Clear Daylight"
	else
		visionMode = "SOFT"
		visionBtn.Text = "🌙 Vision: Soft Fairy Glow"
	end
	if currentState ~= "IDLE" then
		applyNightVision(true)
	end
end)

-- ✨ 3. FAIRY SPARKLES, BEAMS & WAYPOINTS
-- ==========================================
local function addFairySparkles(parent, color)
	local pe = Instance.new("ParticleEmitter")
	pe.Color = ColorSequence.new(color, COLOR_BUTTER_GOLD)
	pe.LightEmission = 1
	pe.Size = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0),
		NumberSequenceKeypoint.new(0.3, 0.25),
		NumberSequenceKeypoint.new(1, 0)
	})
	pe.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 1),
		NumberSequenceKeypoint.new(0.2, 0.1),
		NumberSequenceKeypoint.new(1, 1)
	})
	pe.Lifetime = NumberRange.new(1.2, 2.2)
	pe.Rate = 12
	pe.Speed = NumberRange.new(0.8, 2.0)
	pe.SpreadAngle = Vector2.new(180, 180)
	pe.Parent = parent
	return pe
end

local function createBeam(att0, att1, color, parent)
	local beam = Instance.new("Beam")
	beam.Attachment0 = att0
	beam.Attachment1 = att1
	beam.Color = ColorSequence.new(color, COLOR_BUTTER_GOLD)
	beam.LightEmission = 1
	beam.LightInfluence = 0
	beam.Width0 = 0.45
	beam.Width1 = 0.45
	beam.FaceCamera = true
	beam.Transparency = NumberSequence.new(0.15)
	beam.Parent = parent
	return beam
end

local function addWaypoint(position)
	local node = Instance.new("Part")
	node.Name = "FairyNode_" .. (#waypoints + 1)
	node.Shape = Enum.PartType.Ball
	node.Size = Vector3.new(0.65, 0.65, 0.65)
	node.Position = position
	node.Anchored = true
	node.CanCollide = false
	node.CanQuery = false
	node.CanTouch = false
	node.Material = Enum.Material.Neon
	node.Color = COLOR_EXPLORE
	node.Transparency = 0.15
	node.Parent = trailFolder

	local att = Instance.new("Attachment", node)

	local beam = nil
	if #waypoints > 0 then
		local prevWp = waypoints[#waypoints]
		beam = createBeam(prevWp.attachment, att, COLOR_EXPLORE, node)
	end

	table.insert(waypoints, {
		position = position,
		part = node,
		attachment = att,
		beam = beam
	})

	if wispBeam then
		wispBeam.Attachment0 = att
	end
end

local function removeWaypointsFrom(index, playPop)
	for i = #waypoints, index, -1 do
		if waypoints[i].part then
			waypoints[i].part:Destroy()
		end
		table.remove(waypoints, i)
	end
	if playPop then
		SFX.NodePop()
	end
	if #waypoints > 0 and wispBeam then
		wispBeam.Attachment0 = waypoints[#waypoints].attachment
	end
end

local function cleanupAll()
	currentState = "IDLE"
	if heartbeatConnection then
		heartbeatConnection:Disconnect()
		heartbeatConnection = nil
	end
	if trailFolder then
		trailFolder:Destroy()
		trailFolder = nil
	end
	waypoints = {}
	wispPart = nil
	wispAttachment = nil
	wispBeam = nil
	wispLight = nil
	startBeacon = nil
	distanceLabel = nil

	applyNightVision(false)

	statusLabel.Text = "🌙 Status: Waiting at Entrance"
	statusLabel.TextColor3 = COLOR_WARM_CREAM

	TweenService:Create(startBtn, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(55, 115, 165)}):Play()
	startBtn.Text = "✨ Cast Guiding Thread"

	TweenService:Create(stopBtn, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(52, 45, 68)}):Play()
	stopBtn.TextColor3 = Color3.fromRGB(190, 185, 205)
	stopBtn.Text = "🦋 Follow Thread Home"
end

-- 🧚 4. START & RETURN LOGIC
-- ==========================================
local function startGuide()
	local character = player.Character
	local rootPart = character and character:FindFirstChild("HumanoidRootPart")
	if not rootPart then return end

	cleanupAll()
	SFX.StartMagic()
	currentState = "EXPLORING"

	trailFolder = Instance.new("Folder")
	trailFolder.Name = "PlayerMazeTrail"
	trailFolder.Parent = workspace

	-- Create Floating Fairy Wisp Companion (AC-Safe Proxy Part)
	wispPart = Instance.new("Part")
	wispPart.Name = "FairyWisp"
	wispPart.Shape = Enum.PartType.Ball
	wispPart.Size = Vector3.new(0.9, 0.9, 0.9)
	wispPart.Position = rootPart.Position + Vector3.new(2, 2, 0)
	wispPart.Anchored = true
	wispPart.CanCollide = false
	wispPart.CanQuery = false
	wispPart.CanTouch = false
	wispPart.Material = Enum.Material.Neon
	wispPart.Color = COLOR_BUTTER_GOLD
	wispPart.Parent = trailFolder

	wispAttachment = Instance.new("Attachment", wispPart)
	addFairySparkles(wispPart, COLOR_BUTTER_GOLD)

	wispLight = Instance.new("PointLight")
	wispLight.Color = Color3.fromRGB(255, 238, 200)
	wispLight.Range = 45
	wispLight.Brightness = 1.3
	wispLight.Shadows = false
	wispLight.Parent = wispPart

	applyNightVision(true)

	-- Create Enchanted Start Beacon
	startBeacon = Instance.new("Part")
	startBeacon.Name = "EnchantedStartBeacon"
	startBeacon.Size = Vector3.new(2.2, 6.5, 2.2)
	startBeacon.Position = rootPart.Position
	startBeacon.Anchored = true
	startBeacon.CanCollide = false
	startBeacon.CanQuery = false
	startBeacon.CanTouch = false
	startBeacon.Material = Enum.Material.Neon
	startBeacon.Color = COLOR_EXPLORE
	startBeacon.Transparency = 0.25
	startBeacon.Parent = trailFolder

	local beaconMesh = Instance.new("SpecialMesh", startBeacon)
	beaconMesh.MeshType = Enum.MeshType.FileMesh
	beaconMesh.MeshId = "rbxassetid://9756362" -- Diamond/Crystal shape!
	beaconMesh.Scale = Vector3.new(2.2, 5.5, 2.2)

	addFairySparkles(startBeacon, COLOR_EXPLORE)

	local highlight = Instance.new("Highlight")
	highlight.FillColor = COLOR_EXPLORE
	highlight.OutlineColor = COLOR_BUTTER_GOLD
	highlight.FillTransparency = 0.35
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.Parent = startBeacon

	local billboard = Instance.new("BillboardGui")
	billboard.Size = UDim2.new(0, 170, 0, 50)
	billboard.StudsOffset = Vector3.new(0, 4.5, 0)
	billboard.AlwaysOnTop = true
	billboard.Parent = startBeacon

	distanceLabel = Instance.new("TextLabel")
	distanceLabel.Size = UDim2.new(1, 0, 1, 0)
	distanceLabel.BackgroundTransparency = 1
	distanceLabel.TextColor3 = COLOR_BUTTER_GOLD
	distanceLabel.TextStrokeColor3 = Color3.fromRGB(25, 18, 40)
	distanceLabel.TextStrokeTransparency = 0.1
	distanceLabel.Font = Enum.Font.GothamBold
	distanceLabel.TextSize = 13
	distanceLabel.Text = "✨ SANCTUARY START ✨\n0 studs"
	distanceLabel.Parent = billboard

	addWaypoint(rootPart.Position)
	wispBeam = createBeam(waypoints[1].attachment, wispAttachment, COLOR_EXPLORE, trailFolder)

	-- Update UI Appearance
	TweenService:Create(startBtn, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(45, 85, 125)}):Play()
	startBtn.Text = "🔄 Reset Start Point Here"

	TweenService:Create(stopBtn, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(195, 145, 55)}):Play()
	stopBtn.TextColor3 = Color3.fromRGB(25, 20, 35)
	stopBtn.Text = "🦋 Follow Thread Home"

	local startTime = os.clock()

	heartbeatConnection = RunService.Heartbeat:Connect(function(dt)
		if not rootPart or not rootPart.Parent then
			cleanupAll()
			return
		end

		local t = os.clock() - startTime
		local currentPos = rootPart.Position

		-- Smoothly float the Fairy Wisp beside the player
		if wispPart then
			local hoverOffset = Vector3.new(math.cos(t * 2.2) * 1.8, 1.8 + math.sin(t * 3.5) * 0.4, math.sin(t * 2.2) * 1.8)
			wispPart.Position = wispPart.Position:Lerp(currentPos + hoverOffset, math.clamp(dt * 10, 0, 1))
		end

		-- Gently spin & bob the Start Crystal Beacon
		if startBeacon and waypoints[1] then
			startBeacon.CFrame = CFrame.new(waypoints[1].position + Vector3.new(0, math.sin(t * 2) * 0.6, 0)) * CFrame.Angles(0, t * 1.2, 0)
		end

		local distFromStart = waypoints[1] and math.floor((currentPos - waypoints[1].position).Magnitude) or 0
		if distanceLabel then
			distanceLabel.Text = "✨ SANCTUARY START ✨\n" .. distFromStart .. " studs away"
		end

		if currentState == "EXPLORING" then
			statusLabel.Text = string.format("🦋 Exploring: %d nodes (%dm)", #waypoints, distFromStart)
			statusLabel.TextColor3 = COLOR_EXPLORE

			-- Smart Dead-End Eraser
			if #waypoints > 1 then
				for i = 1, #waypoints - 1 do
					if (currentPos - waypoints[i].position).Magnitude <= BACKTRACK_DISTANCE then
						removeWaypointsFrom(i + 1, false)
						break
					end
				end
			end

			-- Drop new fairy node
			local lastWp = waypoints[#waypoints]
			if lastWp and (currentPos - lastWp.position).Magnitude >= STEP_DISTANCE then
				addWaypoint(currentPos)
			end

		elseif currentState == "RETURNING" then
			statusLabel.Text = string.format("🌟 Guiding Home: %d studs left", distFromStart)
			statusLabel.TextColor3 = COLOR_BUTTER_GOLD

			if #waypoints > 1 then
				local lastWp = waypoints[#waypoints]
				if (currentPos - lastWp.position).Magnitude <= STEP_DISTANCE then
					removeWaypointsFrom(#waypoints, true)
				end
			elseif #waypoints == 1 then
				if (currentPos - waypoints[1].position).Magnitude <= STEP_DISTANCE then
					SFX.StartMagic()
					cleanupAll()
				end
			end
		end
	end)
end

local function stopOrReturn()
	if currentState == "EXPLORING" then
		SFX.ReturnMode()
		currentState = "RETURNING"

		-- Turn the entire thread into glowing Butter Gold to guide the player home
		for _, wp in ipairs(waypoints) do
			if wp.part then wp.part.Color = COLOR_RETURN end
			if wp.beam then
				wp.beam.Color = ColorSequence.new(COLOR_RETURN, Color3.fromRGB(255, 250, 210))
				wp.beam.Width0 = 0.85
				wp.beam.Width1 = 0.85
			end
		end
		if wispBeam then
			wispBeam.Color = ColorSequence.new(COLOR_RETURN, Color3.fromRGB(255, 250, 210))
			wispBeam.Width0 = 0.85
			wispBeam.Width1 = 0.85
		end
		if startBeacon then
			startBeacon.Color = COLOR_RETURN
		end

		TweenService:Create(stopBtn, TweenInfo.new(0.25), {BackgroundColor3 = Color3.fromRGB(185, 65, 85)}):Play()
		stopBtn.TextColor3 = COLOR_WARM_CREAM
		stopBtn.Text = "✖ End & Turn Off Glow"

	elseif currentState == "RETURNING" then
		SFX.StopClean()
		cleanupAll()
	else
		SFX.Click()
	end
end

-- ==========================================
startBtn.MouseButton1Click:Connect(startGuide)
stopBtn.MouseButton1Click:Connect(stopOrReturn)

local charAddedConn = player.CharacterAdded:Connect(function()
	cleanupAll()
end)

	cleanupAll()
	if charAddedConn then charAddedConn:Disconnect() end
	if screenGui then screenGui:Destroy() end
	if soundFolder then soundFolder:Destroy() end
end
